CREATE TABLE TB_FALECONOSCO (
ID_CONTATO INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
NOME_CONTATO VARCHAR (60) NOT NULL,
FONE_CONTATO CHAR (17) NOT NULL,
EMAIL_CONTATO VARCHAR (80) NOT NULL,
ASSUNTO_CONTATO VARCHAR (40) NOT NULL,
MSG_CONTATO TEXT NOT NULL,
RESP_CONTATO TEXT);

CREATE TABLE TB_CADASTRO_ADM (
ID_ADM INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
NOME_ADM VARCHAR (60) NOT NULL,
EMAIL_ADM VARCHAR (50) NOT NULL UNIQUE,
SENHA_ADM VARCHAR (8) NOT NULL);
